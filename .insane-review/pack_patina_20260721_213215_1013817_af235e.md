  1: // Deterministic detection of model-output *leakage* artifacts: tokens that LLM
  2: // web-search / tooling inject and that essentially never appear in human-written
  3: // prose. Unlike the stylometry/lexicon signals (which are probabilistic and
  4: // fire on clusters), a single hit here is near-proof-grade evidence of pasted
  5: // model output, so it fires hard at the document level. See issue #332.
  6: //
  7: // Language-agnostic literal token set — applies to ko/en/zh/ja alike.
  8: //
  9: // Self-scan caveat: patina's own docs, fixtures, and issues that *discuss* these
 10: // tokens will match. That is correct behavior (the text genuinely contains
 11: // them); callers scanning the repo's own meta-content should expect hits.
 12: 
 13: /**
 14:  * Score floor applied when deterministic markup-leakage is detected.
 15:  *
 16:  * Model-output leakage (issue #332) is near-proof-grade: a single token that
 17:  * LLM tooling injects and humans never type. Unlike the stylometric/lexical
 18:  * signals it is decisive on its own, so any hit short-circuits the deterministic
 19:  * `overall` into the 'heavily AI' band (>70) regardless of the per-paragraph
 20:  * hot ratio. It is a floor, not a hard 100, because the surrounding prose may
 21:  * still be genuinely human and we avoid claiming absolute proof.
 22:  *
 23:  * Lives here — the browser-pure module that owns leakage detection — so
 24:  * src/scoring.js re-exports the same constant
 25:  * (threshold parity gate: tests/unit/threshold-parity.test.js).
 26:  *
 27:  * @type {number}
 28:  */
 29: export const LEAKAGE_SCORE_FLOOR = 90;
 30: 
 31: const OBJECT_REPLACEMENT_CHAR = '￼';
 32: 
 33: // Each entry: { id, label, build() => fresh RegExp }. We build a fresh regex per
 34: // scan so the shared module is reentrant (no leaking lastIndex across calls).
 35: const MARKUP_RULES = [
 36:   {
 37:     id: 'oai-citation-markup',
 38:     label: 'OpenAI citation markup',
 39:     build: () => /:contentReference|oaicite|oai_citation/gi,
 40:   },
 41:   {
 42:     id: 'model-tool-token',
 43:     label: 'Model tool token',
 44:     build: () => /\bturn\d+(?:search|view|news|image|forecast|finance|fetch)\d*\b|\bgrok_card\b/gi,
 45:   },
 46:   {
 47:     id: 'object-replacement-char',
 48:     label: 'Object-replacement character (￼)',
 49:     build: () => new RegExp(OBJECT_REPLACEMENT_CHAR, 'g'),
 50:   },
 51:   {
 52:     id: 'ai-tracking-param',
 53:     label: 'AI-tool tracking parameter in URL',
 54:     build: () => /utm_source=(?:chatgpt\.com|openai\.com|perplexity\.ai|claude\.ai|gemini\.google\.com)|[?&](?:ref|utm_source)=chatgpt/gi,
 55:   },
 56:   {
 57:     id: 'explicit-self-identification',
 58:     label: 'Explicit AI self-identification',
 59:     // Near-proof-grade, so precision beats recall (issue #435): a single hit
 60:     // forces LEAKAGE_SCORE_FLOOR. Human bio/ML prose must not fire:
 61:     // - "I am an AI researcher" / "I'm an AI safety engineer": \b alone sits
 62:     //   between "AI" and the next noun, so the bare phrase needs a guard. The
 63:     //   "I am/I'm an AI" alternant (optionally with an AI-role noun:
 64:     //   assistant/chatbot/model) only matches when the phrase ends the clause
 65:     //   (punctuation/EOL) or continues with refusal-boilerplate provenance
 66:     //   ("created/trained/developed ... by"), and never before "-"/"/"
 67:     //   compounds like "AI-powered" or "AI/ML". Human job titles such as
 68:     //   "AI assistant manager" stay cold.
 69:     // - "BERT functions as a language model": the bare alternant now requires
 70:     //   the first-person continuation ("as a language model, I ...") that
 71:     //   real refusal boilerplate uses.
 72:     build: () => /\bas an? (?:AI|artificial intelligence) language model\b|\bas a large language model\b|\bas a language model,? I\b|\bas an AI assistant\b|\bI(?: am|'?m) an AI(?:\s+(?:assistant|chatbot|language model|model))?(?:\s+(?:created|developed|trained|designed|built|made)\s+by\b|(?![-/]|\s+\w))/gi,
 73:   },
 74:   {
 75:     // `navlist` alone is a plausible human identifier (CSS class/id, variable
 76:     // name) in frontend docs/code, so it is NOT near-proof on its own (#442).
 77:     // It only counts as leakage when an independent near-proof token in the
 78:     // same text corroborates that this is pasted model output.
 79:     id: 'model-nav-token',
 80:     label: 'Model navlist token',
 81:     corroborated: true,
 82:     build: () => /\bnavlist\b/gi,
 83:   },
 84: ];
 85: 
 86: /**
 87:  * Scan raw text for model-output leakage artifacts.
 88:  * @param {string} text
 89:  * @returns {{ leaked: boolean, hits: Array<{id:string,label:string,count:number,samples:string[]}> }}
 90:  */
 91: export function detectMarkupLeakage(text) {
 92:   const str = typeof text === 'string' ? text : '';
 93:   const hits = [];
 94:   if (!str) return { leaked: false, hits };
 95: 
 96:   const corroborated = [];
 97:   let hasIndependentHit = false;
 98:   for (const rule of MARKUP_RULES) {
 99:     const matches = str.match(rule.build());
100:     if (!matches || matches.length === 0) continue;
101:     const hit = {
102:       id: rule.id,
103:       label: rule.label,
104:       count: matches.length,
105:       samples: [...new Set(matches.map((m) => m.trim()).filter(Boolean))].slice(0, 3),
106:     };
107:     if (rule.corroborated) {
108:       corroborated.push(hit);
109:     } else {
110:       hits.push(hit);
111:       hasIndependentHit = true;
112:     }
113:   }
114:   // Corroboration-required tokens (navlist) only escalate to leakage when an
115:   // independent near-proof token also fired; otherwise they are dropped so a
116:   // human identifier cannot force the LEAKAGE_SCORE_FLOOR on its own (#442).
117:   if (hasIndependentHit) hits.push(...corroborated);
118:   return { leaked: hits.length > 0, hits };
119: }
120: 
121: export { MARKUP_RULES, OBJECT_REPLACEMENT_CHAR };

  7: # AI-Likeness Scoring Algorithm
  8: 
  9: Pattern-based scoring that converts AI pattern detection results into a numeric 0-100 score.
 10: Used by `--score` mode and `--ouroboros` loop for termination gating.
 11: 
 12: ---
 13: 
 14: ## 1. Severity Scale (Per-Detection)
 15: 
 16: Severity is assigned **per detection** by the LLM during the pattern scan phase.
 17: It is NOT intrinsic to the pattern — the same pattern may receive different severity
 18: depending on how egregiously it appears in context.
 19: 
 20: | Level | Points | Criteria |
 21: |-------|--------|----------|
 22: | High | 3 | Pattern is pervasive — appears multiple times or is especially blatant |
 23: | Medium | 2 | Pattern present at moderate frequency or impact |
 24: | Low | 1 | Pattern barely present — isolated occurrence |
 25: | Not detected | 0 | Pattern not found in text |
 26: 
 27: ---
 28: 
 29: ## 2. Severity Assignment Rubric
 30: 
 31: To reduce variance between runs, follow these guidelines when assigning severity:
 32: 
 33: ### General Rubric (4+ paragraph text)
 34: 
 35: | Instances | Severity |
 36: |-----------|----------|
 37: | 1-2 isolated occurrences | Low (1) |
 38: | 3-5 occurrences, or concentrated in one section | Medium (2) |
 39: | 6+ occurrences, or pervasive throughout | High (3) |
 40: 
 41: ### Special Cases
 42: 
 43: - **Structure patterns (#25-28):** Assess at document level, not instance count.
 44:   A single structural issue (e.g., every paragraph follows identical template) is High.
 45: - **Communication patterns (#19-21):** A single clear chatbot expression
 46:   (e.g., "좋은 질문입니다!") may be High — these are strong AI signals regardless of count.
 47: - **Short text (1-2 paragraphs):** Adjust thresholds proportionally.
 48:   2 instances in 2 paragraphs = Medium, not Low.
 49: 
 50: ---
 51: 
 52: ## 3. Category Derivation
 53: 
 54: Categories are derived **dynamically** from pack frontmatter, not hardcoded.
 55: 
 56: ```

 59:          pack: en-style    → category: style
 60:          pack: ko-custom   → category: custom

 62: Pattern count = pack frontmatter `patterns` field
 63: ```
 64: 
 65: If multiple packs map to the same category (e.g., `ko-content` and `custom/ko-content`),
 66: their patterns are merged and counts summed for that category.

134: Before summing severities, apply profile `pattern-overrides` modifiers:

156:     8: amplify    # ko-language #8 (~적 접미사)
157:     14: suppress  # ko-style #14 (볼드체)
158:   en:
159:     8: amplify    # en-language #8 (Copula Avoidance)
160:     14: suppress  # en-style #14 (Boldface)
161: ```

164: apply **only** that sub-section's overrides. Top-level (unscoped) overrides apply to all
165: languages and are merged before language-scoped ones (language-scoped wins on conflict).

169: ## 6. Scoring Formula
170: 
171: ### Per-Category Score
172: 
173: ```
174: category_score = (sum of adjusted severities / (pattern_count × 3)) × 100
175: ```
176: 
177: - `sum of adjusted severities`: sum severity points for all detected patterns in category,
178:   after applying profile override factors
179: - `pattern_count × 3`: maximum possible score (all patterns detected at High severity)
180: - Result: 0-100 per category
181: 
182: ### Overall Score
183: 
184: ```
185: overall_score = Σ(category_score × category_weight) for all categories
186: ```
187:

243: ## 8. Known Limitations
244: 
245: - **Custom pattern packs** are auto-discovered and scored. A new category
246:   (e.g., `custom/patterns/ko-domain.md` with `pack: ko-domain`) gets default weight 0.10.
247: - **LLM non-determinism** means the same text may score differently across runs.
248:   The formula is deterministic; the severity assignment is not.
249: - **Fidelity scoring** (meaning preservation vs original) is defined in §§ 9–13 below
250:   and integrated into `--score` and `--ouroboros` pipelines.
251: 
252: ### Short-text boost (v3.11 Phase 3.2)
253: 
254: For inputs ≤200 non-whitespace chars OR ≤3 non-empty paragraphs, register-
255: sensitive categories get a 1.5x severity multiplier (capped at 3 per detection):
256: 
257: - `language` — 종결어미, register cues, sentence form patterns
258: - `style` — connectors, transition fillers, formulaic openers
259: - `viral-hook` — shock numbers, clickbait closes, hyperbolic lexicon
260: 
261: Rationale: case-04 found that voice/register shifts (e.g., `~다` ↔ `~습니다`)
262: are clearly perceived by humans but barely move the standard formula because
263: short texts only accumulate 1–2 pattern detections. The boost surfaces those
264: shifts so single-paragraph score deltas align with reader intuition.
265: 
266: The boost is applied at severity-assignment time by the scoring LLM (per the
267: prompt instruction in `buildScoreInstructions`), not as a post-multiplier on
268: category scores. This keeps the formula in §6 unchanged.

272: ## 9. Fidelity Scoring — Overview
273: 
274: AI-likeness (§§ 1–7) measures *how AI-like the output sounds*.
275: Fidelity measures *how faithfully the output preserves the original meaning*.
276: 
277: Both dimensions are necessary: aggressive humanization can achieve a low AI score
278: by deleting content or changing meaning entirely. Fidelity scoring guards against this.
279: 
280: Fidelity scoring is integrated into `--score` and `--ouroboros` pipelines.
281: See SKILL.md § 6 (score mode) for integration details.
282: 
283: ---
284: 
285: ## 10. Fidelity Criteria
286: 
287: Four criteria, each scored independently by the LLM comparing original → output:
288: 
289: ### 10.1 Claims Preserved
290: 
291: Every factual claim in the original appears (perhaps rephrased) in the output.
292: 
293: | Level | Points | Criteria |
294: |-------|--------|----------|
295: | High | 3 | All key claims preserved — no factual content lost |
296: | Medium | 2 | Minor claims omitted — supporting details or examples dropped, but core argument intact |
297: | Low | 1 | Significant claims missing — one or more central facts or arguments absent |
298: | Fail | 0 | Core meaning lost — the output says something fundamentally different |
299: 
300: ### 10.2 No Fabrication
301: 
302: The output does not add claims, facts, or specifics not present or implied by the original.
303: 
304: | Level | Points | Criteria |
305: |-------|--------|----------|
306: | High | 3 | No fabrication — every claim in the output traces to the original |
307: | Medium | 2 | Minor additions — a reasonable inference stated as fact, or an illustrative example added |
308: | Low | 1 | Noticeable fabrication — specific numbers, names, or claims not in the original |
309: | Fail | 0 | Significant fabrication — output contains substantial invented content |
310: 
311: ### 10.3 Tone Match
312: 
313: The output's register matches the original (or the profile's target register, if explicitly overridden).
314: 
315: | Level | Points | Criteria |
316: |-------|--------|----------|
317: | High | 3 | Tone matches — formality level, domain register, and audience consistent |
318: | Medium | 2 | Slight drift — somewhat more/less formal, but still appropriate for the context |
319: | Low | 1 | Noticeable mismatch — formal original made casual (or vice versa) without profile justification |
320: | Fail | 0 | Register violation — academic text made into slang, or casual text made into legalese |
321: 
322: **Profile exception:** When a profile explicitly shifts register (e.g., blog profile amplifies
323: informality), tone match is assessed against the *profile target*, not the original register.
324: 
325: ### 10.4 Length Ratio
326: 
327: Compares output length to original. Extreme changes suggest content loss or padding.
328: 
329: | Ratio | Points | Criteria |
330: |-------|--------|----------|
331: | 70–130% | 3 | Length preserved — natural variation within ±30% |
332: | 50–69% or 131–150% | 2 | Moderate change — some compression or expansion, likely acceptable |
333: | 30–49% or 151–200% | 1 | Significant change — substantial content probably lost or padded |
334: | < 30% or > 200% | 0 | Extreme change — content almost certainly lost or heavily padded |
335: 
336: **Calculation:** `length_ratio = len(output) / len(original) × 100`
337: 
338: Length is measured in characters (not words or tokens) for language-agnostic consistency.
339: 
340: ---
341: 
342: ## 11. Fidelity Severity Assignment Rubric
343: 
344: To reduce variance, apply these guidelines when scoring fidelity criteria:
345: 
346: ### Claims Preserved
347: - Count discrete factual claims in the original. If all appear (rephrased or not) → High.
348: - If only supporting details are dropped but the argument structure survives → Medium.
349: - If a numbered list loses items, a causal chain loses a step, or a key qualifier is dropped → Low.
350: 
351: ### No Fabrication
352: - Paraphrasing that changes word choice but not meaning → High.
353: - Adding a commonly-known context note ("Seoul, the capital of South Korea") → Medium.
354: - Inventing a statistic, date, or name not in the original → Low or Fail.
355: 
356: ### Tone Match
357: - Compare the first and last paragraphs of original vs. output for register cues.
358: - Profile-targeted register shifts are expected, not penalized.
359: - Mixed register (formal opening, casual middle) counts as Low.
360: 
361: ### Length Ratio
362: - This criterion is deterministic — compute the ratio and look up the table.
363: - No LLM judgment needed. Include the raw ratio in the score output.
364:

367: ## 12. Fidelity Scoring Formula
368: 
369: ### Per-Criterion Score
370: 
371: Each criterion is scored 0–3 (same as AI-likeness severity). The fidelity score normalizes
372: across all four criteria:
373: 
374: ```
375: fidelity_score = ((claims + fabrication + tone + length) / 12) × 100
376: ```
377: 
378: - Maximum: (3+3+3+3) / 12 × 100 = **100** (perfect fidelity)
379: - Minimum: (0+0+0+0) / 12 × 100 = **0** (total meaning loss)
380: 
381: ### Criterion Weighting (Optional)
382: 
383: For profiles that need non-uniform criterion importance, weights can be configured:
384: 
385: ```yaml
386: fidelity:
387:   weights:
388:     claims-preserved: 0.35
389:     no-fabrication: 0.30
390:     tone-match: 0.20
391:     length-ratio: 0.15
392: ```
393: 
394: When weights are configured:
395: 
396: ```
397: fidelity_score = Σ((criterion_points / 3) × criterion_weight) × 100
398: ```
399: 
400: Default weights (when not configured): equal at 0.25 each (equivalent to the simple formula).
401: 
402: ### Worked Example
403: 
404: Original: 4-paragraph academic article about climate change policy.
405: Output: Humanized version.
406: 
407: | Criterion | Score | Reasoning |
408: |-----------|-------|-----------|
409: | Claims preserved | 3 (High) | All policy recommendations and cited figures present |
410: | No fabrication | 2 (Medium) | Added "as widely reported" — minor inference stated as fact |
411: | Tone match | 3 (High) | Academic register maintained throughout |
412: | Length ratio | 2 (Moderate) | Output is 68% of original length (within 50-69% band) |
413: 
414: Fidelity = (3+2+3+2) / 12 × 100 = **83.3**
415: 
416: Interpretation: 76-90 range = "높은 충실도" (High fidelity, minor issues)
417: 
418: ### Fidelity Score Interpretation
419: 
420: | Range | Label | Meaning |
421: |-------|-------|---------|
422: | 91-100 | 완벽한 충실도 | Perfect fidelity — all meaning preserved |
423: | 76-90 | 높은 충실도 | High fidelity — minor omissions or additions |
424: | 51-75 | 보통 충실도 | Moderate fidelity — noticeable meaning changes |
425: | 26-50 | 낮은 충실도 | Low fidelity — significant meaning loss or fabrication |
426: | 0-25 | 의미 왜곡 | Meaning severely distorted or lost |
427: 
428: > **Note:** Unlike AI-likeness (lower = better), fidelity uses **higher = better**.
429: > A score of 83 means "roughly 73-93 range" given ±10 LLM variance —
430: > solidly in the "높은 충실도" band.

434: ## 13. Combined Score
435: 
436: ### Formula
437: 
438: AI-likeness and fidelity compose into a combined score with configurable weighting:
439: 
440: ```
441: combined = (ai_likeness × ai_weight) + (fidelity_inverted × fidelity_weight)
442: ```
443: 
444: Where:
445: - `ai_likeness`: AI-likeness score from § 6 (0-100, lower = more human)
446: - `fidelity_inverted`: `100 - fidelity_score` (invert so both dimensions use "lower is better")
447: - `ai_weight + fidelity_weight = 1.0`
448: 
449: ### Default Weights
450: 
451: | Context | AI Weight | Fidelity Weight | Rationale |
452: |---------|-----------|-----------------|-----------|
453: | Default | 0.60 | 0.40 | Balanced — humanization is primary goal |
454: | Academic profile | 0.40 | 0.60 | Meaning preservation is critical in scholarly work |
455: | Blog profile | 0.70 | 0.30 | Creative rewriting tolerated |
456: | Technical profile | 0.35 | 0.65 | Accuracy is paramount in docs |
457: | Social profile | 0.75 | 0.25 | Tone transformation expected |
458: | Email profile | 0.50 | 0.50 | Equal importance |
459: | Legal profile | 0.35 | 0.65 | Legal precision must be preserved |
460: | Medical profile | 0.35 | 0.65 | Clinical accuracy is critical |
461: | Marketing profile | 0.65 | 0.35 | Tone transformation tolerated, creative rewriting expected |
462: | Namuwiki profile | 0.65 | 0.35 | Tone transformation tolerated for wiki-style cleanup |

478: 
479: | Range | Label | Meaning |
480: |-------|-------|---------|
481: | 0-15 | 최적 | Excellent — human-like and faithful |

488:

497: ## 14. MPS (Meaning Preservation Score) — Overview
498: 
499: AI-likeness (§§ 1–7) measures *how AI-like the output sounds*.
500: Fidelity (§§ 9–13) measures *how faithfully the output preserves overall meaning*.
501: MPS measures *whether specific semantic anchors survive the humanization pipeline*.
502: 
503: MPS is anchor-based: it tracks discrete meaning units (claims, polarity, causation, quantifiers, negations) extracted from the original text in SKILL.md Step 4.5, and checks whether each anchor is preserved after each pipeline phase.
504: 
505: MPS complements fidelity scoring — fidelity is a holistic LLM judgment, MPS is a structured anchor-by-anchor verification.
506: 
507: ---
508: 
509: ## 15. MPS Anchor Verification Criteria
510: 
511: Each anchor extracted in Step 4.5 is verified against the pipeline output. Verification produces one of three verdicts:
512: 
513: ### PASS
514: Anchor content is present in the output and its polarity is preserved. The anchor may be rephrased but its core assertion is unambiguously recoverable.
515: 
516: ### SOFT FAIL
517: Anchor content is present but weakened or made ambiguous. Examples:
518: - Specific claim became vague: "매출이 30% 증가" → "매출이 크게 증가"
519: - Quantifier lost precision: "p<0.05" → "통계적으로 유의미하다"
520: - Causal link became correlational: "A 때문에 B가 발생" → "A와 B는 관련이 있다"
521: - Definitive statement became hedged: "시스템이 실패했다" → "시스템에 문제가 있었을 수 있다"
522: 
523: ### HARD FAIL
524: Anchor content is deleted or its polarity is inverted. Examples:
525: - Claim removed entirely from output
526: - Negation dropped: "검증되지 않았다" → "검증되었다"
527: - Causation reversed: "A가 B를 야기했다" → "B가 A를 야기했다"
528: 
529: ### PASS vs SOFT FAIL Boundary
530: 
531: If a reader can **unambiguously recover** the original anchor's meaning from the rewritten version, it is PASS. If a reader could **reasonably interpret** the rewritten version differently from the original, it is SOFT FAIL.
532: 
533: ---
534: 
535: ## 16. MPS Scoring Formula
536: 
537: ### Base Formula
538: 
539: ```
540: anchor_pass_rate = PASS_count / total_anchor_count
541: polarity_preserved = polarity_PASS_count / total_polarity_anchor_count
542: 
543: MPS = (anchor_pass_rate × 0.6 + polarity_preserved × 0.4) × 100
544: ```
545: 
546: Where:
547: - `PASS_count`: anchors with PASS verdict after all remediation (including successful retries)
548: - `total_anchor_count`: all extracted anchors
549: - `polarity_PASS_count`: polarity-type anchors (Polarity + Negation) with PASS verdict
550: - `total_polarity_anchor_count`: all polarity-type anchors (Polarity + Negation)
551: 
552: ### Retry Counting Rule
553: 
554: If a SOFT FAIL anchor passes after alternative correction (retry), it counts as **PASS** in the formula. Only anchors that remain SOFT FAIL or HARD FAIL after all remediation are counted as failures.
555: 
556: ### Fallback (No Polarity Anchors)
557: 
558: If the text contains no Polarity or Negation anchors:
559: 
560: ```
561: MPS = anchor_pass_rate × 100
562: ```
563: 
564: ### Fallback (No Anchors Extracted)
565: 
566: When anchor extraction is skipped (text ≤1 paragraph and ≤2 sentences) or yields zero anchors:
567: 
568: ```
569: MPS = N/A (not applicable)
570: ```
571: 
572: When MPS = N/A:
573: - `--score` mode displays: `의미 보존 (MPS): N/A (앵커 없음)`
574: - Ouroboros loop: MPS floor는 우회되지 않는다. 결정론 루프의 `scoreMPS`는 숫자 mps를 반환하거나 스코어러 실패 시 `null`을 반환하며, `null`은 floor 위반으로 간주해 fail-closed로 직전 반복에 rollback한다 (아래 "Ouroboros Loop Gating" 및 `src/ouroboros.js` 참고). 앵커 N/A 표시는 SKILL 파이프라인 / `--score` 표시용이며 결정론 루프의 floor를 끄지 않는다.
575: 
576: ### MPS Interpretation
577: 
578: | Range | Label | Meaning |
579: |-------|-------|---------|
580: | 90–100 | 우수 (Excellent) | Full meaning preservation |
581: | 70–89 | 양호 (Good) | Minor weakening, acceptable |
582: | 50–69 | 주의 (Warning) | Significant anchor loss, review needed |
583: | < 50 | 위험 (Critical) | Severe meaning corruption |
584: 
585: ### Worked Example
586: 
587: Original text (3 paragraphs) with extracted anchors:
588: 
589: | # | Type | Content | Verdict | After Retry |
590: |---|------|---------|---------|-------------|
591: | 1 | Claim | "시스템이 실패했다" | SOFT FAIL | PASS (retry succeeded) |
592: | 2 | Polarity | "아직 검증되지 않았다" (negative) | PASS | — |
593: | 3 | Quantifier | "매출 30% 증가" | PASS | — |
594: | 4 | Causation | "A 때문에 B 발생" | HARD FAIL | — (original restored) |
595: | 5 | Negation | "불가능하다" | PASS | — |
596: 
597: After remediation:
598: - PASS: #1 (retry), #2, #3, #5 = 4
599: - HARD FAIL: #4 = 1 (original restored, so meaning is preserved in output but pattern not humanized)
600: - Total anchors: 5
601: - Polarity anchors (#2, #5): both PASS = 2/2
602: 
603: ```
604: anchor_pass_rate = 4/5 = 0.80
605: polarity_preserved = 2/2 = 1.00
606: MPS = (0.80 × 0.6 + 1.00 × 0.4) × 100 = (0.48 + 0.40) × 100 = 88
607: ```
608: 
609: Interpretation: 70–89 range = "양호" (Good, minor weakening)
610: 
611: > **Note:** Anchor #4 was a HARD FAIL, so its original sentence was restored in the output.
612: > The anchor counts as a failure in MPS (reducing the score), but the meaning IS preserved
613: > in the output because the original was kept. MPS reflects humanization success rate,
614: > not output meaning accuracy (which is always preserved via fallback).
615: 
616: ### MPS vs Fidelity: Complementary Metrics
617: 
618: MPS measures **humanization coverage** — what fraction of meaning anchors were successfully humanized while being preserved. A HARD FAIL anchor that was restored to its original wording counts as a humanization failure (the pattern wasn't removable without meaning loss), even though the final output's meaning is intact. Fidelity (§§ 9-13) measures **overall output meaning accuracy** against the original — restored sentences score perfectly on fidelity. Use both metrics together: high fidelity + low MPS means "meaning is safe but some AI patterns couldn't be removed."

  72: function matchingBraceEnd(str, open) {
  73:   let depth = 0;

  92:   if (!text) throw new SchemaError('Empty response', text);

  94:   let body = text;
  95:   const codeBlockMatch = text.match(/```(?:json)?\s*([\s\S]*?)\s*```/);

 101:   if (whole.ok && isJsonObject(whole.value)) return whole.value;

 124:       }
 125:     }

 619: /**
 620:  * Reconcile LLM and deterministic overall scores according to config thresholds.
 621:  *
 622:  * @param {object} [options] Reconciliation inputs.
 623:  * @param {number|null} [options.llmOverall] LLM overall score.
 624:  * @param {object|null} [options.deterministicScore] Deterministic score payload.
 625:  * @param {object} [options.config={}] Effective config.
 626:  * @param {object} [options.logger] Logger for warnings.
 627:  * @returns {{overall: number|null, scorePreference: (object|null)}} Reconciled score and preference source.
 628:  * @example
 629:  * const result = reconcileScoreOverall({ llmOverall: 20, deterministicScore: { overall: 60 } });
 630:  */
 631: export function reconcileScoreOverall({
 632:   llmOverall,
 633:   deterministicScore,
 634:   config = {},
 635:   logger,
 636: } = {}) {
 637:   const llm = toFiniteScore(llmOverall);
 638:   const deterministic = toFiniteScore(deterministicScore?.overall);
 639:   if (llm === null) return { overall: null, scorePreference: null };
 640:   if (deterministic === null) return { overall: llm, scorePreference: null };
 641:   // `skipped` means the long-form stylometry/meta block was suppressed (text is
 642:   // ≤2 paragraphs / ≤2 sentences). That must NOT discard the hard evidence
 643:   // floor: near-proof markup leakage (#332) and the structural classifier stay
 644:   // decisive on short text. Enforce that floor here — only the calibrated hard
 645:   // floor, never the coarse per-paragraph hot ratio (1/1 = 100 on one
 646:   // paragraph) — so a short AI-leaked snippet can't be scored 0 just because it
 647:   // was too short for the meta-block, while clean short prose (no hard floor)
 648:   // still defers to the LLM. Non-skipped text keeps the existing divergence path.
 649:   if (deterministicScore?.skipped) {
 650:     const skippedFloor = toFiniteScore(deterministicScore?.evidenceFloor);
 651:     if (skippedFloor !== null && skippedFloor > 0 && llm < skippedFloor) {
 652:       return {
 653:         overall: skippedFloor,
 654:         scorePreference: {
 655:           reason: 'deterministic-evidence-floor',
 656:           selected: 'deterministic',
 657:           llmOverall: llm,
 658:           deterministicOverall: deterministic,
 659:           evidenceFloor: skippedFloor,
 660:           overall: skippedFloor,
 661:         },
 662:       };
 663:     }
 664:     return { overall: llm, scorePreference: null };
 665:   }
 666: 
 667:   const threshold = deterministicScoringOptions(config).divergenceThreshold;
 668:   const delta = Math.abs(llm - deterministic);
 669:   if (delta <= threshold) return { overall: llm, scorePreference: null };
 670: 
 671:   const overall = Math.max(llm, deterministic);
 672:   const selected = overall === deterministic ? 'deterministic' : 'llm';
 673:   const scorePreference = {
 674:     reason: 'deterministic-divergence',
 675:     selected,
 676:     threshold,
 677:     llmOverall: llm,
 678:     deterministicOverall: deterministic,
 679:     overall,
 680:   };
 681:   logger?.warn?.('score.deterministic_divergence', {
 682:     message: `[patina] deterministic score diverged from LLM score (${llm} vs ${deterministic}); using pessimistic ${overall}`,
 683:     llm_overall: llm,
 684:     deterministic_overall: deterministic,
 685:     selected,
 686:   });
 687:   return { overall, scorePreference };
 688: }
