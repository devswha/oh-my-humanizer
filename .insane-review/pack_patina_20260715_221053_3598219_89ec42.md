  2: //
  3: // To avoid duplicated or compounding retries, each resilience concern has
  4: // exactly ONE owner:
  5: //
  6: //   * Transport retry (same provider, same request) — src/api.js `callLLM`.

  8: //     exponential backoff + jitter, bounded by the deadline. CLI backends pass
  9: //     maxRetries=0 (see BACKEND_SAFETY_DEFAULTS) so they never transport-retry.
 10: //   * Backend fallback (different backend) — src/backends/index.js
 11: //     `invokeBackendChain`. On a retryable error it advances to the NEXT backend
 12: //     in the chain; it NEVER re-invokes the same backend (that is transport

178: export async function withBackendConcurrencySlot({
179:   backendName,
180:   maxConcurrency,
181:   signal,
182:   timeout = DEFAULT_BACKEND_TIMEOUT_MS,
183:   deadline = Number.isFinite(timeout) ? Date.now() + timeout : Infinity,
184:   pollMs = 250,
185:   staleMs = Math.max(timeout * 2, 30 * 60_000),
186:   fn,
187: } = {}) {
188:   if (typeof fn !== 'function') {
189:     throw new Error('backend concurrency slot requires fn');
190:   }
191:   // The run phase gets whatever remains of the shared deadline after the slot
192:   // wait, so slot-wait + run can never exceed the single budget (#506 defect 1).
193:   // Callers that pass only `timeout` (no `deadline`) keep their full budget via
194:   // the derived default above.
195:   const remainingTimeout = () =>
196:     (Number.isFinite(deadline) ? Math.max(0, deadline - Date.now()) : timeout);
197:   // Never start backend work on an already-spent budget: an expired shared
198:   // deadline used to reach `fn(0)` on both the unbounded path and via the
199:   // slot-acquired-after-expiry race below (#567).
200:   const assertBudget = () => {
201:     if (Number.isFinite(deadline) && Date.now() >= deadline) {
202:       throw new TimeoutError(`${backendName || 'backend'}: shared deadline expired before backend start`);
203:     }
204:   };
205:   if (!Number.isFinite(maxConcurrency)) {
206:     assertBudget();
207:     return fn(remainingTimeout());
208:   }
209: 
210:   const slot = await acquireBackendSlot({
211:     backendName,
212:     maxConcurrency,
213:     signal,
214:     deadline,
215:     pollMs,
216:     staleMs,
217:   });
218: 
219:   try {
220:     // The slot can be won in the instant the deadline expires (mkdir succeeds
221:     // between the loop's deadline check and now). Re-check before invoking.
222:     assertBudget();
223:     return await fn(remainingTimeout());
224:   } finally {
225:     releaseBackendSlot(slot);
226:   }
227: }

229: async function acquireBackendSlot({
230:   backendName,
231:   maxConcurrency,
232:   signal,
233:   deadline,
234:   pollMs,
235:   staleMs,
236: }) {
237:   throwIfAborted(signal, `${backendName || 'backend'}: aborted while waiting for concurrency slot`);
238:   // Per-user slot root: the slot dirs are real locks, and a world-shared
239:   // tmpdir path means another user's slot dir cannot be removed (EACCES) and
240:   // would permanently consume a cap slot on a multi-user host (#445).
241:   const root = join(tmpdir(), `patina-backend-slots-${userSlotSegment()}`, safePathSegment(backendName || 'backend'));
242:   mkdirSync(root, { recursive: true });
243: 
244:   for (;;) {
245:     // Deadline gates the ROUND, not just the sleep: checking only after a
246:     // failed acquisition round let a slot freed during the sleep be acquired
247:     // after the deadline had already expired (#567).
248:     if (Date.now() >= deadline) {
249:       throw new TimeoutError(`${backendName || 'backend'}: timed out waiting for concurrency slot (cap ${maxConcurrency})`);
250:     }
251:     for (let index = 0; index < maxConcurrency; index++) {
252:       const slot = join(root, `slot-${index}`);
253:       cleanupStaleSlot(slot, staleMs);
254:       try {
255:         mkdirSync(slot);
256:         writeFileSync(join(slot, 'owner.json'), JSON.stringify({
257:           pid: process.pid,
258:           backendName,
259:           createdAt: new Date().toISOString(),
260:         }), 'utf8');
261:         return slot;
262:       } catch (err) {
263:         if (err?.code !== 'EEXIST') throw err;
264:       }
265:     }
266: 
267:     throwIfAborted(signal, `${backendName || 'backend'}: aborted while waiting for concurrency slot`);
268:     if (Date.now() >= deadline) {
269:       throw new TimeoutError(`${backendName || 'backend'}: timed out waiting for concurrency slot (cap ${maxConcurrency})`);
270:     }
271:     await sleepWithAbort(Math.min(pollMs, Math.max(0, deadline - Date.now())), signal, backendName);
272:   }
273: }
274: 
275: function cleanupStaleSlot(slot, staleMs) {
276:   try {
277:     // A crashed owner (its pid no longer alive) must release the slot
278:     // immediately, not after staleMs — otherwise a cap-1 backend (claude/kimi)
279:     // is blocked for up to 30 minutes by a dead run (#445).
280:     if (!isSlotOwnerAlive(slot)) {
281:       rmSync(slot, { recursive: true, force: true });
282:       return;
283:     }
284:     const ageMs = Date.now() - statSync(slot).mtimeMs;
285:     if (ageMs > staleMs) rmSync(slot, { recursive: true, force: true });
286:   } catch {}
287: }
288: 
289: // True unless the slot's recorded owner pid is provably dead. Unreadable/absent
290: // owner records return true so mtime staleness still governs and a just-created
291: // slot (owner.json not yet written) is never yanked from under its owner.
292: function isSlotOwnerAlive(slot) {
293:   let pid;
294:   try {
295:     pid = Number(JSON.parse(readFileSync(join(slot, 'owner.json'), 'utf8'))?.pid);
296:   } catch {
297:     return true;
298:   }
299:   if (!Number.isInteger(pid) || pid <= 0) return true;
300:   try {
301:     process.kill(pid, 0); // signal 0 is a liveness probe, not an actual signal
302:     return true;
303:   } catch (err) {
304:     // ESRCH → no such process (dead); EPERM → exists but not ours (alive).
305:     return err?.code === 'EPERM';
306:   }
307: }
308: 
309: // A filesystem-safe per-user segment so slot roots are owned by, and removable
310: // by, the current user. uid on POSIX; falls back to the username elsewhere.
311: function userSlotSegment() {
312:   try {
313:     const { uid, username } = userInfo();
314:     return Number.isInteger(uid) && uid >= 0 ? `uid-${uid}` : safePathSegment(username || 'user');
315:   } catch {
316:     return 'user';
317:   }
318: }

320: function releaseBackendSlot(slot) {
321:   try {
322:     rmSync(slot, { recursive: true, force: true });
323:   } catch {}
324: }

326: function sleepWithAbort(ms, signal, backendName) {
327:   if (ms <= 0) return Promise.resolve();
328:   if (!signal) return new Promise((resolve) => setTimeout(resolve, ms));
329:   return new Promise((resolve, reject) => {
330:     const timer = setTimeout(() => {
331:       cleanup();
332:       resolve();
333:     }, ms);
334:     const onAbort = () => {
335:       cleanup();
336:       reject(abortError(`${backendName || 'backend'}: aborted while waiting for concurrency slot`));
337:     };
338:     const cleanup = () => {
339:       clearTimeout(timer);
340:       signal.removeEventListener('abort', onAbort);
341:     };
342:     signal.addEventListener('abort', onAbort, { once: true });
343:   });
344: }

347:   if (signal?.aborted) throw abortError(message);
348: }

360: export function runInteractiveCommand({
361:   backendName,
362:   command,
363:   args = [],
364:   cwd = process.cwd(),
365:   env = process.env,
366:   stdio = 'inherit',
367:   notFoundHint,
368: } = {}) {
369:   if (!backendName || !command) {
370:     throw new Error('interactive backend command requires backendName and command');
371:   }
372: 
373:   return new Promise((resolve, reject) => {
374:     // A failed spawn can emit both 'error' and 'close'; settle once so we never
375:     // resolve-then-reject (or build a second Error) on the same invocation (#533).
376:     let settled = false;
377:     const settle = (fn) => { if (settled) return; settled = true; fn(); };
378: 
379:     const proc = spawn(command, args, { cwd, env, stdio });
380:

382:       if (err.code === 'ENOENT') {
383:         settle(() => reject(new Error(`${backendName}: \`${command}\` CLI not found. ${notFoundHint || 'Install the CLI and try again.'}`)));

385:       }
386:       settle(() => reject(new Error(`${backendName}: failed to spawn ${command} (${err.message})`)));
387:     });
388: 
389:     proc.on('close', (code, signal) => {
390:       if (code === 0) {
391:         settle(() => resolve());
392:         return;
393:       }
394:       if (signal) {
395:         settle(() => reject(new Error(`${backendName}: ${command} was terminated by ${signal}`)));
396:         return;

332: // comma-separated list of values for --preview variant comparison. Tokens are

337: function parseTransformList(value, option, valid, hint) {
338:   const tokens = String(value ?? '').split(',').map((t) => t.trim()).filter(Boolean);
339:   if (tokens.length === 0) {
340:     throw inputError(`${option} expects a value`, `Valid values are: ${valid.join(', ')}.`, hint);
341:   }
342:   const seen = [];
343:   for (const token of tokens) {
344:     if (!valid.includes(token)) {
345:       throw inputError(
346:         `unknown ${TRANSFORM_OPTION_NOUNS[option] ?? 'value'} ${token}`,
347:         `Valid values are: ${valid.join(', ')}.`,
348:         hint
349:       );
350:     }
351:     if (!seen.includes(token)) seen.push(token);
352:   }
353:   return seen.join(',');
354: }

357:   return String(value ?? fallback).split(',').map((t) => t.trim()).filter(Boolean);

360: // Expand --jargon/--tone into the rewrite variants a run executes.
361: // A single combination is the normal one-call path; multiple combinations
362: // (comma lists) become --preview compare variants, one rewrite call each.
363: // Tone joins the cross product so "remove in casual" and "remove in marketing"
364: // are directly comparable; tone appears in the label only when it varies.
365: // The cross product is capped: every variant is a full LLM call.
366: export const MAX_TRANSFORM_VARIANTS = 4;
367: 
368: export function buildTransformVariants(parsed) {
369:   const jargons = splitTransformValues(parsed.jargon, 'keep');
370:   const tones = splitTransformValues(parsed.tone, '');
371:   const toneList = tones.length > 0 ? tones : [null];
372:   const multiTone = toneList.length > 1;
373:   const variants = [];
374:   for (const jargon of jargons) {
375:     for (const tone of toneList) {
376:       const parts = [];
377:       if (jargon !== 'keep') parts.push(jargon);
378:       let label = parts.join('+');
379:       if (multiTone) label = label ? `${label}·${tone}` : tone;
380:       variants.push({ jargon, tone, label: label || 'cleanup' });
381:     }
382:   }
383:   if (variants.length > MAX_TRANSFORM_VARIANTS) {
384:     throw inputError(
385:       `too many transform variants (${variants.length})`,
386:       `--jargon × --tone combinations are capped at ${MAX_TRANSFORM_VARIANTS}; each variant is a full rewrite call.`,
387:       'Drop values from one of the lists, e.g. `--jargon keep,remove` with a single --tone.'
388:     );
389:   }
390:   return variants;
391: }

546: // Output routing flags are batch-only and mutually exclusive. Without this,
547: // `patina --in-place draft.md` silently prints to stdout (never overwriting),
548: // combined destinations apply hidden precedence, and --outdir can collapse
549: // distinct inputs onto one output file (#440).
550: export function validateOutputRouting(parsed) {
551:   const destinations = [
552:     parsed.inPlace ? '--in-place' : null,
553:     parsed.suffix !== undefined ? '--suffix' : null,
554:     parsed.outdir !== undefined ? '--outdir' : null,
555:   ].filter(Boolean);
556:   if (destinations.length === 0) return;
557:   if (!parsed.batch && !parsed.xliff) {
558:     throw inputError(
559:       `${destinations[0]} requires --batch`,
560:       'Output routing flags only apply to batch mode; without --batch the result goes to stdout.',
561:       `Run \`patina --batch ${destinations[0]}${destinations[0] === '--in-place' ? '' : ' <value>'} <files>\`.`
562:     );
563:   }
564:   if (destinations.length > 1) {
565:     throw inputError(
566:       `${destinations[0]} and ${destinations[1]} cannot be combined`,
567:       'Each batch run writes to exactly one destination; combining them would silently pick one.',
568:       'Pick one of --in-place, --suffix, or --outdir.'
569:     );
570:   }
571:   // An empty value (e.g. `--suffix=` or `--outdir=`) passes the
572:   // `!== undefined` destination check above but is falsy, so writeBatchOutput
573:   // would silently fall through to stdout (#504). Reject it here so the
574:   // validator's "present" guarantee matches writeBatchOutput's truthiness gate.
575:   if (parsed.suffix === '') {
576:     throw inputError(
577:       '--suffix requires a non-empty value',
578:       'An empty --suffix= would leave the filename unchanged and silently print to stdout instead of writing files.',
579:       'Pass a suffix like --suffix .patina (writes draft.patina.md).'
580:     );
581:   }
582:   if (parsed.outdir === '') {
583:     throw inputError(
584:       '--outdir requires a non-empty value',
585:       'An empty --outdir= would silently print to stdout instead of writing files.',
586:       'Pass a directory like --outdir out/.'
587:     );
588:   }
589:   if (parsed.outdir !== undefined) {
590:     const seen = new Map();
591:     for (const file of parsed.files) {
592:       const base = basename(file);
593:       const prior = seen.get(base);
594:       if (prior !== undefined && prior !== file) {
595:         throw inputError(
596:           `--outdir would overwrite ${base}`,
597:           `Both "${prior}" and "${file}" map to the same output file in ${parsed.outdir}.`,
598:           'Rename the inputs, or use --suffix / --in-place to keep outputs beside their sources.'
599:         );
600:       }
601:       seen.set(base, file);
602:     }
603:   }
604: }

606: // XLIFF mode is an explicit, file-to-file localization pass: it humanizes
607: // already-translated <target> segments through the normal rewrite+verify path,
608: // so it rejects non-rewrite modes, meaning/register-changing flags, and stdin —
609: // anything that would make byte-preserving localization unsafe. Run this BEFORE
610: // the generic validators so the errors are XLIFF-specific.
611: export function validateXliffRequest(parsed) {
612:   if (!parsed.xliff) {
613:     // --dry-run and --max-segments are XLIFF-only in this MVP.
614:     if (parsed.dryRun) {
615:       throw inputError('--dry-run requires --xliff',
616:         '--dry-run only reports an XLIFF humanize plan; it has no meaning for the normal rewrite modes.',
617:         'Run `patina --xliff --dry-run <file.xliff>`.');
618:     }
619:     if (parsed.maxSegments !== undefined) {
620:       throw inputError('--max-segments requires --xliff',
621:         '--max-segments only caps the XLIFF humanize segment count.',
622:         'Run `patina --xliff --max-segments <n> <file.xliff>`.');
623:     }
624:     return;
625:   }
626: 
627:   /** @type {[string, string][]} incompatible mode/flag pairs (parsed key, flag). */
628:   const incompatible = [
629:     ['audit', '--audit'], ['score', '--score'], ['diff', '--diff'],
630:     ['preview', '--preview'], ['ocr', '--ocr'], ['serve', '--serve'],
631:     ['gate', '--exit-on'], ['persona', '--persona'], ['jargon', '--jargon'],
632:     ['tone', '--tone'], ['profile', '--profile'], ['rewriteHeadings', '--rewrite-headings'],
633:   ];
634:   for (const [key, flag] of incompatible) {
635:     if (parsed[key] !== undefined && parsed[key] !== false) {
636:       throw inputError(`${flag} cannot be combined with --xliff`,
637:         `--xliff is a meaning-preserving localization pass over translated <target> segments; ${flag} is not part of that surface in this MVP.`,
638:         `Drop ${flag}, or run it separately without --xliff.`);
639:     }
640:   }
641:   if (parsed.verify) {
642:     throw inputError('--verify cannot be combined with --xliff',
643:       'XLIFF mode always verifies each rewritten segment against the MPS/fidelity floors; verification cannot be disabled or toggled.',
644:       'Drop --verify — it is implied by --xliff.');
645:   }
646:   if (!parsed.files || parsed.files.length === 0) {
647:     throw inputError('--xliff requires file paths, not stdin',
648:       'XLIFF humanize reads and rewrites structured files in place, so it needs at least one .xliff file argument.',
649:       'Run `patina --xliff <file.xliff>`.');
650:   }
651:   if (parsed.files.length > 1 && !parsed.batch) {
652:     throw inputError('--xliff with multiple files requires --batch',
653:       'Each XLIFF run writes one output file; processing several inputs at once is batch mode.',
654:       'Run `patina --batch --xliff <file1.xliff> <file2.xliff>`.');
655:   }
656: }

701: function parseFailureRateOption(value, option) {
702:   const n = numericOptionValue(value);
703:   if (!Number.isFinite(n) || n < 0) {
704:     throw inputError(
705:       `${option} expects a ratio or percent`,
706:       `Received ${value === undefined ? 'no value' : `"${value}"`}.`,
707:       `Use ${option} 0.25 for 25%, or ${option} 25.`
708:     );
709:   }
710:   const ratio = n > 1 ? n / 100 : n;
711:   if (ratio > 1) {
712:     throw inputError(
713:       `${option} expects a value from 0 to 1 or 0 to 100`,
714:       `Received "${value}".`,
715:       `Use ${option} 0.25 for 25%, or ${option} 25.`
716:     );
717:   }
718:   // A value in the open (1, 2) interval is ambiguous: it is too big to be a
719:   // ratio (>1) so it is read as a percent, but a tiny one (1.5 -> 1.5% -> 0.015)
720:   // almost certainly is not what the user meant (#508 G4). Stay backward
721:   // compatible (no throw) but surface the interpretation so the mistake is
722:   // visible instead of silent.
723:   if (n > 1 && n < 2) {
724:     process.stderr.write(`[patina] --max-failure-rate ${value} read as ${n}% (ratio ${ratio}). Use a value <=1 for a ratio (0.015 = 1.5%) or >=2 for a clear percent.\n`);
725:   }
726:   return ratio;
727: }

 604: /**
 605:  * Create a SIGINT-aware cancellation controller for long-running CLI operations.
 606:  *
 607:  * @param {object} [options] Cancellation integration points.
 608:  * @param {NodeJS.Process} [options.processObj=process] Process-like object used for signal listeners.
 609:  * @param {NodeJS.WritableStream} [options.stderr=process.stderr] Stream for fallback cancel messages.
 610:  * @param {object|null} [options.logger] Optional patina logger.
 611:  * @returns {{signal: AbortSignal, install: Function, cleanup: Function, throwIfCanceled: Function}} Controller facade.
 612:  * @example
 613:  * const cancellation = createCancellationController();
 614:  * cancellation.install();
 615:  */
 616: 
 617: export function createCancellationController({
 618:   processObj = process,
 619:   stderr = process.stderr,
 620:   logger = null,
 621: } = {}) {
 622:   const controller = new AbortController();
 623:   let sigintCount = 0;
 624:   let installed = false;
 625: 
 626:   const writeStatus = (message) => {
 627:     if (logger) {
 628:       logger.warn('cli.cancel', { message: message.trimEnd() });
 629:       return;
 630:     }
 631:     if (stderr && typeof stderr.write === 'function') stderr.write(message);
 632:   };
 633: 
 634:   const onSigint = () => {
 635:     sigintCount++;
 636:     if (sigintCount === 1) {
 637:       processObj.exitCode = 130;
 638:       writeStatus('[patina] cancelling… press Ctrl-C again to exit immediately\n');
 639:       controller.abort();
 640:       return;
 641:     }
 642: 
 643:     cleanup();
 644:     processObj.exit(130);
 645:   };
 646: 
 647:   function install() {
 648:     if (!installed && typeof processObj.on === 'function') {
 649:       processObj.on('SIGINT', onSigint);
 650:       installed = true;
 651:     }
 652:   }
 653: 
 654:   function cleanup() {
 655:     if (installed && typeof processObj.removeListener === 'function') {
 656:       processObj.removeListener('SIGINT', onSigint);
 657:       installed = false;
 658:     }
 659:   }
 660: 
 661:   return {
 662:     signal: controller.signal,
 663:     install,
 664:     cleanup,
 665:     throwIfCanceled() {
 666:       if (controller.signal.aborted) throw cancellationError();
 667:     },
 668:   };
 669: }

 715: async function runPreviewJob({
 716:   parsed,
 717:   config,
 718:   patterns,
 719:   profile,
 720:   voice,
 721:   scoring,
 722:   toneResolution,
 723:   promptMode,
 724:   backends,
 725:   resolved,
 726:   repoRoot,
 727:   timeoutMs,
 728:   logger,
 729: }) {
 730:   const input = parsed.files[0];
 731:   const isUrl = /^https?:\/\//i.test(String(input));
 732:   const cancellation = createCancellationController({ logger });
 733:   cancellation.install();
 734:   try {
 735:     let pageHtml = null;
 736:     let blocks = null;
 737:     let sourceUrl = null;
 738:     let sourcePath = input;
 739:     let originalText;
 740:     let snapshotSource = null;
 741: 
 742:     if (isUrl) {
 743:       logger.info('preview.fetch', { message: `[patina] Fetching ${input}` });
 744:       let page;
 745:       try {
 746:         page = await fetchPreviewPage(input, { signal: cancellation.signal, timeoutMs });
 747:       } catch (err) {
 748:         throw runtimeError(
 749:           'could not fetch the preview page',
 750:           `${input}: ${err?.message || 'fetch failed'}`,
 751:           'Check the URL is reachable from this machine, or save the page HTML to a file and run `patina --preview file.html`.'
 752:         );
 753:       }
 754:       cancellation.throwIfCanceled();
 755:       snapshotSource = page.html;
 756:       sourceUrl = page.finalUrl;
 757:     } else {
 758:       const [loaded] = await loadInputs(parsed, logger);
 759:       sourcePath = loaded.path;
 760:       // Local files are validated to .html upstream and use the same
 761:       // snapshot pipeline as a fetched page.
 762:       snapshotSource = loaded.text;
 763:       sourceUrl = pathToFileURL(resolve(process.cwd(), sourcePath)).href;
 764:     }
 765: 
 766:     if (snapshotSource !== null) {
 767:       pageHtml = prepareSnapshotHtml(snapshotSource);
 768:       if (isUrl) {
 769:         // Must happen before extraction: inlining changes offsets, and the
 770:         // in-place swap later relies on the block offsets captured here.
 771:         pageHtml = await freezeSnapshotAssets(pageHtml, {
 772:           baseUrl: sourceUrl,
 773:           signal: cancellation.signal,
 774:           logger,
 775:         });
 776:         cancellation.throwIfCanceled();
 777:       }
 778:       const extracted = extractProseBlocks(pageHtml);
 779:       blocks = extracted.blocks;
 780:       // With --ocr, a page whose copy lives entirely in images has no DOM
 781:       // prose but is exactly the case OCR exists for — defer the no-prose
 782:       // error until after OCR has had a chance to find image text.
 783:       if (blocks.length === 0 && !parsed.ocr) {
 784:         throw runtimeError(
 785:           'no prose found on the page',
 786:           'The page has no plain-text prose blocks patina can rewrite in place (often a client-rendered SPA, or text split by inline markup).',
 787:           'Try a server-rendered page, save the article text to a file, or add --ocr to scan image text.'
 788:         );
 789:       }
 790:       if (extracted.truncated) {
 791:         logger.warn('preview.truncated', {
 792:           message: '[patina] Page has more prose blocks than the preview limit; extra blocks are left unchanged.',
 793:         });
 794:       }
 795:       originalText = blocks.map((block) => block.text).join('\n\n');
 796:       if (blocks.length > 0) {
 797:         logger.info('preview.blocks', {
 798:           message: `[patina] Rewriting ${blocks.length} prose block(s) from ${sourceUrl}`,
 799:         });
 800:       }
 801:     }
 802: 
 803:     const basePromptInputs = {
 804:       config,
 805:       patterns,
 806:       profile: profile.body ? profile : null,
 807:       voice: voice.body ? voice : null,
 808:       scoring: scoring.body ? scoring : null,
 809:       tone: toneResolution,
 810:       promptMode,
 811:       jargon: parsed.jargon,
 812:       rewriteHeadings: parsed.rewriteHeadings,
 813:     };
 814:     const invokeInputs = {
 815:       backends,
 816:       apiKey: resolved.apiKey,
 817:       baseURL: resolved.baseURL,
 818:       model: resolved.model,
 819:       modelSource: resolved.modelSource,
 820:       signal: cancellation.signal,
 821:       timeout: timeoutMs,
 822:       maxConcurrency: parsed.maxConcurrency,
 823:       maxRetries: parsed.maxRetries,
 824:       logger,
 825:     };
 826: 
 827:     // --ocr: extract text from page images and let it ride the same rewrite
 828:     // call as extra paragraph blocks. Image text cannot be swapped back into
 829:     // pixels, so changed findings render as annotations + notes cards.
 830:     let ocrImages = [];
 831:     if (parsed.ocr) {
 832:       if (pageHtml === null) {
 833:         logger.warn('ocr.skipped', {
 834:           message: '[patina] --ocr applies to URL/.html previews; plain-text input has no images.',
 835:         });
 836:       } else {
 837:         ocrImages = await runOcrStage({
 838:           pageHtml,
 839:           sourceUrl,
 840:           parsed,
 841:           backends,
 842:           resolved,
 843:           timeoutMs,
 844:           cancellation,
 845:           logger,
 846:         });
 847:       }
 848:     }
 849: 
 850:     // A page with no DOM prose AND no image text has nothing to rewrite.
 851:     if (blocks !== null && blocks.length === 0 && ocrImages.length === 0) {
 852:       throw runtimeError(
 853:         'no prose found on the page',
 854:         'The page has no plain-text prose blocks, and --ocr found no text in its images.',
 855:         'Try a server-rendered page, or save the page HTML to a file and run `patina --preview file.html`.'
 856:       );
 857:     }
 858: 
 859:     const rewriteText = [originalText, ...ocrImages.map((image) => image.text)]
 860:       .filter(Boolean)
 861:       .join('\n\n');
 862:     const documentContext = buildDocumentSignals({ text: rewriteText, lang: config.language || 'ko' });
 863: 
 864:     // Variant comparison (--jargon x,y / --tone a,b): one rewrite call per
 865:     // variant, all baked into the preview page behind a scriptless toggle.
 866:     // Calls run sequentially — local CLI backends carry concurrency caps of
 867:     // 1-2, and a variant is a whole-document rewrite, not a cheap request.
 868:     const transformVariants = buildTransformVariants(parsed);
 869:     const compareMode = transformVariants.length > 1;
 870:     if (compareMode && pageHtml === null) {
 871:       throw runtimeError(
 872:         'transform-variant comparison needs a page snapshot',
 873:         'Plain-text file previews render as a single reading document, which cannot hold multiple toggleable variants.',
 874:         'Run the compare against a URL or .html input, or pick a single --jargon/--tone value.'
 875:       );
 876:     }
 877:     const variantBodies = [];
 878:     let rewrittenBody;
 879:     if (compareMode) {
 880:       const previewLang = config.language || 'ko';
 881:       const firstCliTone = typeof parsed.tone === 'string' ? parsed.tone.split(',')[0] : parsed.tone;
 882:       for (const [index, variant] of transformVariants.entries()) {
 883:         logger.info('preview.variant', {
 884:           message: `[patina] Rewriting variant ${variant.label} (${index + 1}/${transformVariants.length})…`,
 885:         });
 886:         // Per-variant tone: a comma-listed --tone makes each variant carry its
 887:         // own register, re-resolved here to mirror a single run with that --tone.
 888:         // Profile (genre) is fixed across variants — tone no longer remaps it.
 889:         let variantTone = toneResolution;
 890:         if (variant.tone && variant.tone !== firstCliTone) {
 891:           variantTone = resolveTone({ cliTone: variant.tone, configTone: config.tone, lang: previewLang });
 892:           if (variantTone.warning) {
 893:             logger.warn('tone.warning', { message: `[patina] ${variantTone.warning}` });
 894:           }
 895:         }
 896:         const variantProfile = basePromptInputs.profile;
 897:         const variantRaw = await invokeBackendChain({
 898:           ...invokeInputs,
 899:           prompt: buildPrompt({
 900:             ...basePromptInputs,
 901:             profile: variantProfile,
 902:             tone: variantTone,
 903:             jargon: variant.jargon,
 904:             text: rewriteText,
 905:             mode: 'rewrite',
 906:             documentSignals: documentContext.signals,
 907:           }),
 908:         });
 909:         cancellation.throwIfCanceled();
 910:         variantBodies.push(formatRewriteBodyForBrowser(variantRaw, { logger }));
 911:       }
 912:       rewrittenBody = variantBodies[0];
 913:     } else {
 914:       const rawResult = await invokeBackendChain({
 915:         ...invokeInputs,
 916:         prompt: buildPrompt({
 917:           ...basePromptInputs,
 918:           text: rewriteText,
 919:           mode: 'rewrite',
 920:           documentSignals: documentContext.signals,
 921:         }),
 922:       });
 923:       cancellation.throwIfCanceled();
 924:       rewrittenBody = formatRewriteBodyForBrowser(rawResult, { logger });
 925:     }
 926: 
 927:     // Best-effort pattern explanation, same contract as the browser diff
 928:     // page: one extra call, and a failure never fails the preview. Compare
 929:     // mode skips it: one explanation per variant would multiply the call
 930:     // budget, and the variant toggle itself is the comparison surface.
 931:     let explanationHtml = '';
 932:     if (compareMode) {
 933:       logger.info('preview.variant_explanation_skipped', {
 934:         message: '[patina] explanation call skipped in compare mode (one rewrite call per variant already).',
 935:       });
 936:     } else try {
 937:       const diffResult = await invokeBackendChain({
 938:         ...invokeInputs,
 939:         prompt: buildPrompt({
 940:           ...basePromptInputs,
 941:           text: buildBrowserDiffPromptInput(rewriteText, rewrittenBody),
 942:           mode: 'diff',
 943:         }),
 944:       });
 945:       const explanation = formatOutput(
 946:         diffResult,
 947:         'diff',
 948:         { ...parsed, format: 'markdown', noColor: true },
 949:         { logger, stdout: { isTTY: false } },
 950:       );
 951:       explanationHtml = renderExplanationHtml(explanation);
 952:     } catch (err) {
 953:       logger.warn('preview.diff_failed', {
 954:         message: `[patina] preview explanation failed: ${err?.message || 'diff call failed'}`,
 955:       });
 956:     }
 957:     cancellation.throwIfCanceled();
 958: 
 959:     // Score symmetric scopes: with --ocr the rewrite covers DOM text + image
 960:     // text, so the "before" must too (rewriteText), or the chip would compare
 961:     // unequal scopes and misreport the change. Compare mode scores every
 962:     // variant so the chip shows where each one lands.
 963:     const beforeScore = scoreDeterministicSignals({ text: rewriteText, config, repoRoot, logger });
 964:     let scoreChip = null;
 965:     if (!beforeScore?.skipped && beforeScore?.overall !== null && beforeScore?.overall !== undefined) {
 966:       if (compareMode) {
 967:         const parts = transformVariants.map((variant, index) => {
 968:           const variantScore = scoreDeterministicSignals({ text: variantBodies[index], config, repoRoot, logger });
 969:           return !variantScore?.skipped && variantScore?.overall !== null && variantScore?.overall !== undefined
 970:             ? `${variant.label} ${variantScore.overall}`
 971:             : null;
 972:         }).filter(Boolean);
 973:         scoreChip = parts.length > 0 ? `score ${beforeScore.overall} → ${parts.join(' · ')}` : null;
 974:       } else {
 975:         const afterScore = scoreDeterministicSignals({ text: rewrittenBody, config, repoRoot, logger });
 976:         scoreChip = !afterScore?.skipped && afterScore?.overall !== null && afterScore?.overall !== undefined
 977:           ? `score ${beforeScore.overall} → ${afterScore.overall}`
 978:           : null;
 979:       }
 980:     }
 981: 
 982:     let built;
 983:     let stdoutBody = rewrittenBody;
 984:     if (pageHtml !== null) {
 985:       // Align each rewrite body against the extracted blocks independently —
 986:       // models merge/split paragraphs differently per variant.
 987:       const alignOne = (body, label) => {
 988:         try {
 989:           const aligned = alignRewrites([...blocks, ...ocrImages], body);
 990:           if (aligned.unalignedCount > 0) {
 991:             logger.warn('preview.partial_alignment', {
 992:               message: `[patina] ${aligned.unalignedCount} block(s)${label ? ` in variant ${label}` : ''} could not be aligned with the rewrite and keep their original text.`,
 993:             });
 994:           }
 995:           return aligned.rewrites;
 996:         } catch (err) {
 997:           throw runtimeError(
 998:             'preview rewrite could not be aligned',
 999:             `${err.message}, so the rewrites cannot be swapped back into the page safely.`,
1000:             'Re-run the command (model output varies), or save the page HTML to a file and run `patina --preview file.html`.'
1001:           );
1002:         }
1003:       };
1004:       const rewrites = alignOne(rewrittenBody, compareMode ? transformVariants[0].label : '');
1005:       const previewVariants = compareMode
1006:         ? transformVariants.map((variant, index) => ({
1007:           label: variant.label,
1008:           jargon: variant.jargon,
1009:           tone: variant.tone,
1010:           rewrites: (index === 0 ? rewrites : alignOne(variantBodies[index], variant.label)).slice(0, blocks.length),
1011:         }))
1012:         : null;
1013:       const imageFindings = ocrImages.map((image, index) => {
1014:         const rewritten = rewrites[blocks.length + index];
1015:         return { ...image, rewritten, changed: rewritten !== image.text };
1016:       });
1017:       if (ocrImages.length > 0) {
1018:         // Keep stdout pipe-safe: only the page's own text, never OCR blocks.
1019:         stdoutBody = rewrites.slice(0, blocks.length).join('\n\n');
1020:       }
1021:       built = buildPreviewHtml({
1022:         html: pageHtml,
1023:         blocks,
1024:         rewrites: rewrites.slice(0, blocks.length),
1025:         variants: previewVariants,
1026:         sourceUrl,
1027:         explanationHtml,
1028:         scoreChip,
1029:         imageFindings,
1030:         contextCardHtml: buildContextCardHtml({
1031:           register: documentContext.register,
1032:           // With per-variant tones one global tone row would be wrong for
1033:           // every variant but the first — show the register measurement only.
1034:           tone: compareMode && transformVariants.some((v) => v.tone !== transformVariants[0].tone)
1035:             ? null
1036:             : toneResolution,
1037:         }),
1038:       });
1039:       if (compareMode) {
1040:         logger.info('preview.variants_ready', {
1041:           message: `[patina] ${transformVariants.length} variants baked in (${transformVariants.map((v) => v.label).join(', ')}); stdout carries "${transformVariants[0].label}". Toggle variants from the preview bar.`,
1042:         });
1043:       }
1044:     }
1045:     // Do the throwing/binding work (temp-file write, serve port bind) BEFORE the
1046:     // large stdout write: process.exit() does not drain a piped stdout, so a

147: ### Personas (voice)
148: 
149: A **persona** is a reusable voice — a built-in (`patina persona list`) or your own, authored without editing source:
150: 
151: ```bash
152: patina persona new my-voice --from-sample past-posts.txt   # learn from your writing
153: patina persona new my-voice --describe "plain-spoken founder, casual"
154: patina --persona my-voice draft.md                          # then reuse it
155: 
156: patina persona show my-voice                                # inspect the normalized config (--json for machine output)
157: patina persona edit my-voice --name "Founder voice"         # copy-on-edit into custom/personas/ (built-ins stay intact)
158: patina persona rm my-voice                                  # remove a custom persona (--force to skip the confirm)
159: ```
160: 
161: Works on ko/en/zh/ja and composes with `--tone` (register) and `--profile` (pattern policy). The persona is the sole voice owner; register precedence is `--tone` > persona. A persona shapes voice but never lowers the meaning floors — authored personas are validated on save, and the safety gate still enforces MPS/fidelity + dropped-number checks.

191: 
192: ```text
193: Input
194:   -> semantic anchor extraction (claims, polarity, causation, numbers)
195:   -> stylometry + AI-lexicon scan
196:   -> pattern-guided rewrite
